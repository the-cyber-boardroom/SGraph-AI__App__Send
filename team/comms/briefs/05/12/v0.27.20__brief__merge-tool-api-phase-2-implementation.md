## Step 3 — Extend `vault-tool-api.js` with Phase 2 Methods

Add these methods in `vault-tool-api.js`, all registered via `api.register(name, fn, { async, events })`. Place after the four Phase 1 registrations.

### File Operations (mirror `sg.vfs.*`)

```js
api.register('readFile', async ({ path }) => {
    if (!path) throw new Error('readFile: path required')
    const shell = window.sgraphVault?.shell
    if (!shell?._dataSource) throw new Error('readFile: vault not ready')
    return await shell._dataSource.getFileBytes(path)   // ArrayBuffer
}, { async: true })

api.register('writeFile', async ({ path, content }) => {
    if (!path) throw new Error('writeFile: path required')
    const shell = window.sgraphVault?.shell
    if (!shell?._dataSource?.writable) throw new Error('writeFile: read-only vault')
    // path = '/folder/file.txt' → folder='/folder', file='file.txt'
    const slash = path.lastIndexOf('/')
    const folder = slash > 0 ? path.slice(0, slash) : '/'
    const file   = path.slice(slash + 1)
    let bytes
    if (typeof content === 'string') {
        bytes = new TextEncoder().encode(content).buffer
    } else if (content instanceof ArrayBuffer) {
        bytes = content
    } else if (content instanceof Uint8Array) {
        bytes = content.buffer
    } else {
        throw new Error('writeFile: content must be string | ArrayBuffer | Uint8Array')
    }
    await shell._dataSource.saveFile(folder, file, bytes)
    // Auto-push to keep named ref in sync (same behaviour as sg.vfs.write)
    if (shell._vault?.push) {
        try { await shell._vault.push() } catch (err) { console.warn('[vault-tool-api] auto-push failed:', err) }
    }
    _dispatchVaultEvent(VAULT_EVENTS.FILE_SAVED, { path })
    _dispatchStateChanged({ change: 'file-saved', path })
    return { path, size: bytes.byteLength }
}, { async: true, events: [VAULT_EVENTS.FILE_SAVED] })

api.register('listFiles', ({ path } = {}) => {
    const shell = window.sgraphVault?.shell
    if (!shell?._dataSource) return []
    const all = shell._dataSource.getFileList()
    const prefix = (path || '').replace(/^\//, '')
    return prefix ? all.filter(f => f.path.startsWith(prefix)) : all
}, { async: false })

api.register('deleteFile', async ({ path }) => {
    if (!path) throw new Error('deleteFile: path required')
    const shell = window.sgraphVault?.shell
    if (!shell?._dataSource?.writable) throw new Error('deleteFile: read-only vault')
    const slash = path.lastIndexOf('/')
    const folder = slash > 0 ? path.slice(0, slash) : '/'
    const file   = path.slice(slash + 1)
    await shell._dataSource.deleteFile(folder, file)
    if (shell._vault?.push) { try { await shell._vault.push() } catch (_) {} }
    _dispatchStateChanged({ change: 'file-deleted', path })
    return { deleted: true, path }
}, { async: true })

api.register('createFolder', async ({ path }) => {
    if (!path) throw new Error('createFolder: path required')
    const shell = window.sgraphVault?.shell
    if (!shell?._dataSource?.writable) throw new Error('createFolder: read-only vault')
    await shell._dataSource.createFolder(path)
    if (shell._vault?.push) { try { await shell._vault.push() } catch (_) {} }
    _dispatchStateChanged({ change: 'folder-created', path })
    return { created: true, path }
}, { async: true })
```

### Sync Operations (mirror `sg.git.*`)

```js
api.register('push', async () => {
    const shell = window.sgraphVault?.shell
    if (!shell?._vault || !shell._dataSource?.writable) throw new Error('push: read-only vault')
    await shell._vault.push()
    shell._refreshSyncState()
    _dispatchStateChanged({ change: 'pushed' })
    return { pushed: true }
}, { async: true })

api.register('pull', async () => {
    const shell = window.sgraphVault?.shell
    if (!shell?._vault || !shell._dataSource?.writable) throw new Error('pull: read-only vault')
    const changed = await shell._vault.pull()
    if (changed) {
        await shell._mountBrowse()
        shell._refreshSyncState()
    }
    _dispatchStateChanged({ change: 'pulled', changed: !!changed })
    return { pulled: !!changed }
}, { async: true })

api.register('checkUpstream', async () => {
    const shell = window.sgraphVault?.shell
    if (!shell?._checkBehindOnly) throw new Error('checkUpstream: shell not ready')
    await shell._checkBehindOnly(false)
    const s = shell._syncState || {}
    return { ahead: s.ahead || 0, behind: s.behind || 0, diverged: !!s.diverged }
}, { async: true, events: [VAULT_EVENTS.SYNC_CHANGED] })
```

### Auth (mirror `sg.auth.*`)

```js
api.register('setAccessKey', async ({ key }) => {
    const shell = window.sgraphVault?.shell
    if (!shell) throw new Error('setAccessKey: shell not ready')
    const trimmed = String(key || '').trim()
    // Route through the same event that the settings page and inline unlock panel use
    shell.dispatchEvent(new CustomEvent('vault-settings-access-key', {
        bubbles: true, detail: { key: trimmed }
    }))
    if (!trimmed) return { ok: true, valid: false, remaining: 0 }
    // Validate server-side (does not consume a use)
    const endpoint = shell._vault?._sgSend?.endpoint || window.location.origin
    try {
        const r = await fetch(`${endpoint}/api/transfers/check_token/${encodeURIComponent(trimmed)}`)
        const d = await r.json()
        return { ok: true, valid: !!d.valid, remaining: d.remaining, status: d.status }
    } catch (_) {
        return { ok: true, valid: null }
    }
}, { async: true })

api.register('checkAccessKey', async ({ key }) => {
    const trimmed = String(key || '').trim()
    if (!trimmed) return { valid: false, remaining: 0 }
    const shell = window.sgraphVault?.shell
    const endpoint = shell?._vault?._sgSend?.endpoint || window.location.origin
    const r = await fetch(`${endpoint}/api/transfers/check_token/${encodeURIComponent(trimmed)}`)
    const d = await r.json()
    return { valid: !!d.valid, remaining: d.remaining, status: d.status }
}, { async: true })

api.register('clearAccessKey', () => {
    const shell = window.sgraphVault?.shell
    if (!shell) throw new Error('clearAccessKey: shell not ready')
    shell.dispatchEvent(new CustomEvent('vault-settings-access-key', {
        bubbles: true, detail: { key: '' }
    }))
    return { cleared: true }
}, { async: false })
```

## Step 4 — Update `vault-events.js`

Add two new event constants (keep the existing four):

```js
export const VAULT_EVENTS = Object.freeze({
    OPENED:              'vault:opened',
    LOCKED:              'vault:locked',
    NAVIGATION_COMPLETE: 'vault:navigation-complete',
    FILE_SAVED:          'vault:file-saved',
    SYNC_CHANGED:        'vault:sync-changed',     // NEW
    FILE_DELETED:        'vault:file-deleted',     // NEW
})
```

In `vault-tool-api.js`, also wire `_dispatchVaultEvent(VAULT_EVENTS.SYNC_CHANGED, ...)` to fire whenever `shell._refreshSyncState()` updates ahead/behind/diverged. The cleanest hook: monkey-patch `shell._refreshSyncState` once during `initVaultToolApi()`:

```js
const _origRefreshSync = shell._refreshSyncState.bind(shell)
shell._refreshSyncState = async function() {
    const before = JSON.stringify(shell._syncState)
    await _origRefreshSync()
    const after  = JSON.stringify(shell._syncState)
    if (before !== after) {
        _dispatchVaultEvent(VAULT_EVENTS.SYNC_CHANGED, shell._syncState)
        _dispatchStateChanged({ change: 'sync-changed', state: shell._syncState })
    }
}
```

This is safe because `_refreshSyncState` is called from a small set of well-defined spots in vault-shell.js.
